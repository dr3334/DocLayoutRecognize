cmake_minimum_required(VERSION 3.22)
# 与 APK 的 Debug build type 独立；在 project()/NDK 初始化前确定优化类型。
set(CMAKE_BUILD_TYPE Release CACHE STRING "生产原生引擎使用 Release 优化" FORCE)
project(android_dococr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ANDROID_OCR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(DOCOCR_SOURCE_ROOT "${ANDROID_OCR_ROOT}/native/docprase" CACHE PATH "docprase 源码快照")
if(NOT EXISTS "${DOCOCR_SOURCE_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "仓库内缺少 native/docprase 源码快照，请重新获取完整项目")
endif()
option(MNN_USE_PREBUILT "默认使用仓库内预编译 MNN，无需下载或编译 MNN" ON)
set(DOCOCR_REQUIRE_MNN ON CACHE BOOL "必须包含真实视觉后端" FORCE)
set(DOCOCR_REQUIRE_LLM ON CACHE BOOL "必须包含真实 LLM 后端" FORCE)
set(DOCOCR_BUILD_TESTS OFF CACHE BOOL "Android 不构建桌面测试" FORCE)
if(MNN_USE_PREBUILT)
  include("${CMAKE_CURRENT_LIST_DIR}/mnn_prebuilt.cmake")
else()
  set(MNN_SOURCE_ROOT "${ANDROID_OCR_ROOT}/native/MNN" CACHE PATH "维护者指定的固定 MNN 源码")
  if(NOT EXISTS "${MNN_SOURCE_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "源码模式缺少 MNN，请运行 python tools/setup_native_deps.py --source")
  endif()
  include("${CMAKE_CURRENT_LIST_DIR}/mnn_source.cmake")
endif()
add_subdirectory("${DOCOCR_SOURCE_ROOT}" docprase EXCLUDE_FROM_ALL)
include("${CMAKE_CURRENT_LIST_DIR}/android_engine_sources.cmake")
# Android 宿主自身使用日志 API，不依赖 MNN 源码 target 间接传递。
find_library(android_log_library log REQUIRED)
target_link_libraries(dococr_c PRIVATE ${android_log_library})
# 桌面 PDF 渲染器依赖外部 Poppler/posix_spawn；保留 Android 原有 Java PDF 路径。
get_target_property(core_sources dococr_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "(^|/)pdf_renderer\\.cpp$")
set_property(TARGET dococr_core PROPERTY SOURCES "${core_sources}")
target_sources(dococr_core PRIVATE "${CMAKE_CURRENT_LIST_DIR}/pdf_renderer_android.cpp")
target_include_directories(dococr_core PRIVATE "${DOCOCR_SOURCE_ROOT}/src")
add_library(dococr_jni SHARED dococr_jni.cpp)
target_include_directories(dococr_jni PRIVATE "${DOCOCR_SOURCE_ROOT}/third_party/stb")
target_link_libraries(dococr_jni PRIVATE dococr_c)
target_compile_features(dococr_jni PRIVATE cxx_std_17)
